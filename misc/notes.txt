#!/bin/bash
set -euo pipefail

BASEDIR="$( cd "$( dirname "${BASH_SOURCE[0]}" )" && pwd )"
cd $BASEDIR

source .env
#VAULT_PATH="${VAULT_PATH:-ssh_keys/mykey}"

# Generate key if it does not exist
if [[ ! -f "$SSH_KEY_NAME" ]]; then
    ssh-keygen -t ed25519 -f "$SSH_KEY_NAME" -C "crossplane" -N ""
fi

# Read keys into variables
PRIVATE_KEY=$(<"$SSH_KEY_NAME")
PUBLIC_KEY=$(<"${SSH_KEY_NAME}.pub")

# Store keys in Vault
vault kv put "$VAULT_PATH" private_key="$PRIVATE_KEY" public_key="$PUBLIC_KEY"

---

vault secrets list

---

if diff <(ssh-keygen -y -f id_ed25519_crossplane) id_ed25519_crossplane.pub >/dev/null; then
    echo "Keys match"
else
    echo "Keys do NOT match"
fi
